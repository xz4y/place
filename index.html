-- ========================================================= -- PLACE profiles 설정 -- ========================================================= -- --------------------------------------------------------- -- 1. profiles 기본 테이블 -- --------------------------------------------------------- create table if not exists profiles ( id uuid references auth.users on delete cascade primary key, username text unique not null, nickname text not null, created_at timestamptz default timezone('utc'::text, now()) not null, nickname_updated_at timestamptz default timezone('utc'::text, now()) not null ); -- --------------------------------------------------------- -- 2. 필요한 컬럼 추가 -- --------------------------------------------------------- alter table profiles add column if not exists account_number bigint; alter table profiles add column if not exists is_admin boolean default false; alter table profiles add column if not exists ban_type text default null; alter table profiles add column if not exists ban_until timestamptz default null; alter table profiles add column if not exists ban_reason text default null; alter table profiles add column if not exists chatlock_until timestamptz default null; alter table profiles add column if not exists chatlock_reason text default null; -- --------------------------------------------------------- -- 3. 기존 account_number의 Identity 제거 -- --------------------------------------------------------- alter table profiles alter column account_number drop identity if exists; -- --------------------------------------------------------- -- 4. 기존 계정 번호 정리 -- --------------------------------------------------------- with numbered as ( select id, row_number() over ( order by created_at, id ) as new_number from profiles ) update profiles p set account_number = numbered.new_number from numbered where p.id = numbered.id; -- --------------------------------------------------------- -- 5. 가입번호 자동 증가용 Sequence -- --------------------------------------------------------- create sequence if not exists profiles_account_number_seq; -- 현재 가장 큰 번호로 Sequence 맞추기 select setval( 'profiles_account_number_seq', greatest( coalesce( (select max(account_number) from profiles), 0 ), 1 ) ); -- 앞으로 가입하는 계정은 자동으로 번호 증가 alter table profiles alter column account_number set default nextval('profiles_account_number_seq'); alter sequence profiles_account_number_seq owned by profiles.account_number; -- --------------------------------------------------------- -- 6. 관리자 계정 설정 -- --------------------------------------------------------- -- 모든 계정을 일단 일반 유저로 설정 update profiles set is_admin = false; -- place 계정만 관리자 update profiles set is_admin = true where lower(username) = 'place'; -- --------------------------------------------------------- -- 7. profiles RLS -- --------------------------------------------------------- alter table profiles enable row level security; drop policy if exists "Public profiles read" on profiles; drop policy if exists "Users insert own profile" on profiles; drop policy if exists "Users update own profile" on profiles; -- 누구나 프로필 읽기 가능 create policy "Public profiles read" on profiles for select to anon, authenticated using (true); -- 본인 프로필 생성 가능 create policy "Users insert own profile" on profiles for insert to authenticated with check ( auth.uid() = id ); -- 본인 프로필 수정 가능 create policy "Users update own profile" on profiles for update to authenticated using ( auth.uid() = id ) with check ( auth.uid() = id ); -- ========================================================= -- CHATS -- ========================================================= create table if not exists chats ( id bigint generated by default as identity primary key, username text not null, message text not null, channel text default 'all', map text, user_id uuid, sender_username text, target_username text, created_at timestamptz default now() ); -- 필요한 컬럼이 없으면 추가 alter table chats add column if not exists channel text default 'all'; alter table chats add column if not exists map text; alter table chats add column if not exists user_id uuid; alter table chats add column if not exists sender_username text; alter table chats add column if not exists target_username text; -- --------------------------------------------------------- -- chats RLS -- --------------------------------------------------------- alter table chats enable row level security; drop policy if exists "Enable read access for all users" on chats; create policy "Enable read access for all users" on chats for select to anon, authenticated using (true); drop policy if exists "Enable insert access for all users" on chats; drop policy if exists "Enable insert for authenticated users only" on chats; create policy "Enable insert for authenticated users only" on chats for insert to authenticated with check ( auth.role() = 'authenticated' );
